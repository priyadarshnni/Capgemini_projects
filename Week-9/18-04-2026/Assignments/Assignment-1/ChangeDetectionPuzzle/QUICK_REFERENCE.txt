import { Component, Input, ChangeDetectionStrategy, ChangeDetectorRef } from '@angular/core';

@Component({
  selector: 'app-dashboard',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `Score: {{ userStats.score }}`
})
export class DashboardComponent {
  @Input() userStats!: UserStats;

  constructor(private cdr: ChangeDetectorRef) {}

  updateLocally() {
    this.userStats.score = 100;
    this.cdr.markForCheck();  // ← Schedule check in next cycle
  }
}

SOLUTION 3: Immutable Pattern (Object Spread)
═══════════════════════════════════════════════════════════════════

import { Component, Input, ChangeDetectionStrategy } from '@angular/core';

@Component({
  selector: 'app-dashboard',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `Score: {{ userStats.score }}`
})
export class DashboardComponent {
  @Input() userStats!: UserStats;

  // NO ChangeDetectorRef needed!
  
  updateLocally() {
    // Create NEW object with updated value
    this.userStats = {
      ...this.userStats,         // Copy all properties
      score: 100                 // Override score
    };
    // ✓ View updates automatically (new reference!)
  }
}

IMMUTABLE PATTERN BEHAVIOR:
┌──────────────────────────────────┐
│ OLD: this.userStats.score = 100  │
│ (same reference)                 │
│ userStats ─────→ { score: 100 }  │
│            address #12345        │
│ ❌ OnPush doesn't detect!        │
└──────────────────────────────────┘

┌──────────────────────────────────┐
│ NEW: this.userStats = {           │
│   ...this.userStats,             │
│   score: 100                     │
│ }                                │
│ (new reference!)                 │
│ userStats ─────→ { score: 100 }  │
│            address #54321        │
│ ✓ OnPush detects change!         │
└──────────────────────────────────┘

EXAMPLE 1: Single field update
══════════════════════════════
// Immutable (best):
updateScore(newScore: number) {
  this.userStats = { ...this.userStats, score: newScore };
}

// Or detectChanges if you must mutate:
updateScore(newScore: number) {
  this.userStats.score = newScore;
  this.cdr.detectChanges();
}


EXAMPLE 2: Nested object update
═════════════════════════════════
// OLD (wrong - doesn't work with OnPush):
updateAchievements(newAchlevements: string[]) {
  this.userStats.achievements = newAchievements;
  // ❌ Same reference, OnPush won't detect
}

// CORRECT (immutable):
updateAchievements(newAchievements: string[]) {
  this.userStats = {
    ...this.userStats,           // Copy all
    achievements: newAchievements // Override nested
  };
  // ✓ New reference detected by OnPush
}

// CORRECT (if you must mutate):
updateAchievements(newAchievements: string[]) {
  this.userStats.achievements = newAchievements;
  this.cdr.detectChanges();
}


EXAMPLE 3: Array operations with nested updates
════════════════════════════════════════════════

// Add achievement - IMMUTABLE WAY:
addAchievement(achievement: string) {
  this.userStats = {
    ...this.userStats,
    achievements: [...this.userStats.achievements, achievement]
  };
}

// Remove achievement - IMMUTABLE WAY:
removeAchievement(index: number) {
  this.userStats = {
    ...this.userStats,
    achievements: this.userStats.achievements.filter((_, i) => i !== index)
  };
}


EXAMPLE 4: API response handling
═════════════════════════════════
// Best practice:
loadUserData() {
  this.api.getUserStats().subscribe(stats => {
    this.userStats = stats;  // New reference from API
    // ✓ Automatically detected by OnPush
  });
}
